```python
import json
import os

# 步骤 1: 读取JSON文件
def read_json_file(file_path):
    with open(file_path, 'r', encoding='utf-8') as file:  # 指定编码为UTF-8
        data = json.load(file)
    return data

# 步骤 2: 根据ID区间筛选信息
def filter_messages_by_id(data, start_id, end_id):
    filtered_messages = [message for message in data['messages'] if start_id <= message['id'] <= end_id]
    return filtered_messages

# 步骤 3: 导出到JSON文件
def export_to_json(data, file_name):
    with open(file_name, 'w') as file:
        json.dump(data, file, indent=4)

# 步骤 4: 自定义文件夹名字并保存文件
def save_to_custom_folder(folder_name, file_name, data):
    if not os.path.exists(folder_name):
        os.makedirs(folder_name)
    file_path = os.path.join(folder_name, file_name)
    export_to_json(data, file_path)

# 步骤 5: 主程序
def main():
    json_file_path = 'tdl-export.json'
    data = read_json_file(json_file_path)
    
    start_id = int(input("请输入起始ID: "))
    end_id = int(input("请输入结束ID: "))
    folder_name = input("请输入要保存的文件夹名字: ")
    
    filtered_messages = filter_messages_by_id(data, start_id, end_id)
    file_name = f'tdl-export-{start_id}-{end_id}.json'
    
    save_to_custom_folder(folder_name, file_name, {"messages": filtered_messages})
    print(f'文件已保存到 {folder_name}/{file_name}')

if __name__ == '__main__':
    main()
```
