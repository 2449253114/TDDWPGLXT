以下是一个示例，说明如何根据你的代码返回动态HTML内容：

    // ...你的支付逻辑...

    // 生成HTML内容
    const htmlContent = `
        <!DOCTYPE html>
        <html lang="zh-CN">
        <head>
            <meta charset="UTF-8">
            <title>支付成功</title>
        </head>
        <body>
            <h1>支付成功</h1>
            <span>订单编号：${queryParam.out_trade_no}</span>
        </body>
        </html>
    `;

    return {
        code: 0, // 假设0代表成功
        data: htmlContent,
        msg: '支付成功'
    };
}
